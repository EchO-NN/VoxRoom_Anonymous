cmake_minimum_required(VERSION 3.16)
project(voxroom_robot)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(octomap REQUIRED)
find_package(octomap_msgs REQUIRED)
find_package(pacecat_m_series_inter REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
add_executable(m200_input_bridge src/m200_input_bridge.cpp)
target_include_directories(m200_input_bridge PRIVATE include)
ament_target_dependencies(m200_input_bridge rclcpp sensor_msgs std_msgs pacecat_m_series_inter livox_ros_driver2)
add_executable(body_to_lidar src/body_to_lidar.cpp)
target_include_directories(body_to_lidar PRIVATE include)
ament_target_dependencies(body_to_lidar rclcpp sensor_msgs geometry_msgs tf2 tf2_ros)
add_executable(octomap_snapshot src/octomap_snapshot.cpp)
target_include_directories(octomap_snapshot PRIVATE ${OCTOMAP_INCLUDE_DIRS})
ament_target_dependencies(octomap_snapshot rclcpp octomap_msgs std_msgs tf2 tf2_ros)
target_link_libraries(octomap_snapshot ${OCTOMAP_LIBRARIES})
add_executable(octomap_export_voxroom src/octomap_export_voxroom.cpp)
target_include_directories(octomap_export_voxroom PRIVATE ${OCTOMAP_INCLUDE_DIRS})
target_link_libraries(octomap_export_voxroom ${OCTOMAP_LIBRARIES})
install(TARGETS m200_input_bridge body_to_lidar octomap_snapshot octomap_export_voxroom DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS ../../../run_online.py DESTINATION lib/${PROJECT_NAME})
install(FILES ../../../snapshot_adapter.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_core test/test_core.cpp)
  target_include_directories(test_core PRIVATE include)
  add_test(NAME test_core COMMAND test_core)
endif()
ament_package()
